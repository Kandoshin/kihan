def normalize_tuple(x, y, w, h, W, H):
    # x, y, w, h 是上面算出来的绝对数值
    # W, H 是游戏画面的总宽高
    return (round(x / W, 6), round(y / H, 6), round(w / W, 6), round(h / H, 6))
