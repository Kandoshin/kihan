# ============= 键盘监听器类 =============
class KeyboardListener:
    """监听数字键输入来切换训练模式"""

    def __init__(self):
        self.running = False
        self.thread = None

    def start(self):
        """启动键盘监听"""
        if not self.running:
            self.running = True
            self.thread = threading.Thread(target=self._listen_loop, daemon=True)
            self.thread.start()
            print("【键盘监听】已启动 - 按数字键0/1/2切换训练模式")

    def stop(self):
        """停止键盘监听"""
        self.running = False
        if self.thread:
            self.thread.join()

    def _listen_loop(self):
        """监听循环"""
        global TRAINING_MODE
        last_check = 0

        while self.running:
            try:
                current_time = time.time()
                if current_time - last_check > 0.1:  # 100ms检查一次
                    for key in ['0', '1', '2']:
                        if keyboard.is_pressed(key):
                            new_mode = int(key)
                            if new_mode != TRAINING_MODE:
                                TRAINING_MODE = new_mode
                                mode_names = ["AI自主学习", "人类示教", "离线数据训练"]
                                print(f"\n🔄 【模式切换】切换到模式 {TRAINING_MODE}: {mode_names[TRAINING_MODE]}")
                                time.sleep(0.5)  # 防止重复触发
                    last_check = current_time
                time.sleep(0.01)
            except Exception as e:
                print(f"键盘监听错误: {e}")
                time.sleep(0.1)


# ============= 人类示教数据管理器 =============
class HumanDataManager:
    """管理人类示教数据的保存和加载"""

    def __init__(self, data_dir=HUMAN_DATA_DIR):
        self.data_dir = data_dir
        self.current_session = []
        self.session_start_time = None

    def start_session(self):
        """开始新的示教会话"""
        self.current_session = []
        self.session_start_time = datetime.now()
        print(f"📚 【人类示教】开始记录数据 - {self.session_start_time.strftime('%H:%M:%S')}")

    def add_experience(self, obs, action, reward, next_obs, done):
        """添加一条经验数据"""
        experience = {
            'obs': obs.copy(),
            'action': action.copy(),
            'reward': reward,
            'next_obs': next_obs.copy(),
            'done': done,
            'timestamp': time.time()
        }
        self.current_session.append(experience)

    def save_session(self):
        """保存当前会话数据"""
        if not self.current_session:
            print("⚠️ 【人类示教】没有数据需要保存")
            return None

        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"human_demo_{timestamp}.pkl"
        filepath = os.path.join(self.data_dir, filename)

        session_data = {
            'experiences': self.current_session,
            'session_info': {
                'start_time': self.session_start_time,
                'end_time': datetime.now(),
                'num_experiences': len(self.current_session),
                'total_reward': sum(exp['reward'] for exp in self.current_session)
            }
        }

        try:
            with open(filepath, 'wb') as f:
                pickle.dump(session_data, f)
            print(f"💾 【人类示教】数据已保存: {filename} ({len(self.current_session)} 条经验)")
            self.current_session = []
            return filepath
        except Exception as e:
            print(f"❌ 【人类示教】保存失败: {e}")
            return None

    def load_all_data(self):
        """加载所有历史示教数据"""
        all_experiences = []
        files = [f for f in os.listdir(self.data_dir) if f.startswith('human_demo_') and f.endswith('.pkl')]

        for filename in files:
            try:
                filepath = os.path.join(self.data_dir, filename)
                with open(filepath, 'rb') as f:
                    session_data = pickle.load(f)
                all_experiences.extend(session_data['experiences'])
                print(f"📖 加载数据文件: {filename} ({len(session_data['experiences'])} 条)")
            except Exception as e:
                print(f"❌ 加载文件失败 {filename}: {e}")

        print(f"📊 总共加载了 {len(all_experiences)} 条人类示教经验")
        return all_experiences


# ============= 离线训练器 =============
class OfflineTrainer:
    """使用人类示教数据进行离线训练"""

    def __init__(self, model, data_manager, learning_rate=OFFLINE_LEARNING_RATE):
        self.model = model
        self.data_manager = data_manager
        self.learning_rate = learning_rate

    def train_from_human_data(self, num_epochs=OFFLINE_EPOCHS, batch_size=OFFLINE_BATCH_SIZE):
        """从人类数据训练"""
        # 加载所有人类示教数据
        experiences = self.data_manager.load_all_data()
        if not experiences:
            print("❌ 【离线训练】没有找到人类示教数据")
            return False

        print(f"🎯 【离线训练】开始训练，数据量: {len(experiences)}, 批次大小: {batch_size}, 轮数: {num_epochs}")

        # 准备数据
        obs_data = np.array([exp['obs'] for exp in experiences])
        action_data = np.array([exp['action'] for exp in experiences])
        reward_data = np.array([exp['reward'] for exp in experiences])

        # 转换为torch tensors
        obs_tensor = torch.FloatTensor(obs_data).to(DEVICE)
        action_tensor = torch.FloatTensor(action_data).to(DEVICE)

        # 设置优化器
        optimizer = torch.optim.Adam(self.model.policy.parameters(), lr=self.learning_rate)

        total_loss = 0
        num_batches = 0

        for epoch in range(num_epochs):
            epoch_loss = 0
            indices = np.random.permutation(len(experiences))

            for i in range(0, len(experiences), batch_size):
                batch_indices = indices[i:i + batch_size]
                batch_obs = obs_tensor[batch_indices]
                batch_actions = action_tensor[batch_indices]

                # 计算行为克隆损失
                with torch.no_grad():
                    # 获取策略分布
                    features = self.model.policy.extract_features(batch_obs)
                    latent_pi = self.model.policy.mlp_extractor.forward_actor(features)
                    action_logits = self.model.policy.action_net(latent_pi)

                # 计算交叉熵损失（行为克隆）
                optimizer.zero_grad()

                # 重新计算 logits（需要梯度）
                features = self.model.policy.extract_features(batch_obs)
                latent_pi = self.model.policy.mlp_extractor.forward_actor(features)
                action_logits = self.model.policy.action_net(latent_pi)

                # 二元交叉熵损失
                loss = F.binary_cross_entropy_with_logits(action_logits, batch_actions)

                loss.backward()
                optimizer.step()

                epoch_loss += loss.item()
                num_batches += 1

            avg_epoch_loss = epoch_loss / (len(experiences) // batch_size + 1)
            total_loss += avg_epoch_loss
            print(f"📈 Epoch {epoch + 1}/{num_epochs}, Loss: {avg_epoch_loss:.4f}")

        avg_total_loss = total_loss / num_epochs
        print(f"✅ 【离线训练】完成! 平均损失: {avg_total_loss:.4f}")
        return True


# ============= CNN 处理器 (原有代码保持不变) =============
def init_cnn_model():
    """在当前进程中初始化 CNN 模型"""
    try:
        model = models.mobilenet_v3_small(weights=None)
        model.classifier[3] = nn.Linear(model.classifier[3].in_features, num_classes)

        if os.path.exists(cnn_weight_path):
            sd = torch.load(cnn_weight_path, map_location=DEVICE)
            model.load_state_dict(sd, strict=True)
            print(f"成功加载 CNN 权重: {cnn_weight_path}")
        else:
            print(f"警告：找不到 CNN 权重文件 {cnn_weight_path}")

        model.to(DEVICE).eval()

        embedding_layer = nn.Embedding(num_classes, 16).to(DEVICE)
        transform = transforms.Compose([
            transforms.Resize((img_size, img_size)),
            transforms.ToTensor(),
            transforms.Normalize([0.5], [0.5])
        ])
        return model, embedding_layer, transform
    except Exception as e:
        print(f"CNN 模型初始化失败: {e}")
        return None, None, None


class ThreadCNNProcessor:
    """使用线程池而非多进程来避免句柄问题"""

    def __init__(self, max_workers=2):
        self.max_workers = max_workers
        self.executor = None
        self.cnn_model = None
        self.embedding_layer = None
        self.transform = None

    def start(self):
        """启动线程池并初始化模型"""
        global TRAINING_MODE
        if TRAINING_MODE == 2:  # 离线训练模式不需要CNN
            print("🔄 【离线模式】跳过CNN初始化")
            return True

        try:
            self.cnn_model, self.embedding_layer, self.transform = init_cnn_model()
            if self.cnn_model is None:
                print("警告：CNN 模型初始化失败，将使用默认值")
                return False

            self.executor = ThreadPoolExecutor(max_workers=self.max_workers)
            print(f"CNN 处理器启动成功，使用 {self.max_workers} 个线程")
            return True
        except Exception as e:
            print(f"CNN 处理器启动失败: {e}")
            return False

    def stop(self):
        """关闭线程池"""
        if self.executor:
            self.executor.shutdown(wait=True)
            print("CNN 处理器已关闭")

    def convert_predictions_to_features(self, pred_ids):
        """转换预测结果为特征"""
        try:
            cnn_vecs = []
            for i, pred_id in enumerate(pred_ids):
                if i >= len(cnn_regions):
                    break

                label = idx_to_class.get(int(pred_id), "none")
                region_name = cnn_regions[i]["name"]

                if region_name in {"1p", "2p"}:
                    if label == "none":
                        cnn_vecs.extend([0.0] * 16)
                    else:
                        vec = self.embedding_layer(torch.tensor([pred_id], device=DEVICE)).squeeze(
                            0).detach().cpu().numpy()
                        cnn_vecs.extend(vec.tolist())
                elif region_name in {"stand", "skill1", "skill2"}:
                    cnn_vecs.append(int(label) / 60.0 if label.isdigit() else 0.0)
                elif region_name in {"summon", "scroll"}:
                    if label.isdigit():
                        val = int(label) / 60.0
                        cnn_vecs.extend([val] * 16)
                    elif label == "none":
                        cnn_vecs.extend([0.0] * 16)
                    else:
                        vec = self.embedding_layer(torch.tensor([pred_id], device=DEVICE)).squeeze(
                            0).detach().cpu().numpy()
                        cnn_vecs.extend(vec.tolist())
                elif region_name in {"1p_energy", "2p_energy"}:
                    cnn_vecs.append(int(label) / 4.0 if label.isdigit() else 0.0)
                elif region_name == "time":
                    if label == "none":
                        cnn_vecs.append(-1.0)
                    elif label.isdigit():
                        val = int(label)
                        cnn_vecs.append(round(val / 60.0, 4))
                    else:
                        cnn_vecs.append(-1.0)
            return cnn_vecs
        except Exception as e:
            print(f"转换预测结果失败: {e}")
            return [0.0] * self.get_expected_dim()

    def get_expected_dim(self):
        """计算预期的特征维度"""
        dim = 0
        for reg in cnn_regions:
            name = reg["name"]
            if name in {"1p", "2p"}:
                dim += 16
            elif name in {"summon", "scroll"}:
                dim += 16
            elif name in {"stand", "skill1", "skill2"}:
                dim += 1
            elif name in {"1p_energy", "2p_energy"}:
                dim += 1
            elif name == "time":
                dim += 1
        return dim

    def predict(self, crop_images, timeout=2):
        """预测图像特征"""
        if self.cnn_model is None or self.executor is None:
            return [0.0] * self.get_expected_dim()

        try:
            future = self.executor.submit(self._predict_sync, crop_images)
            return future.result(timeout=timeout)
        except Exception as e:
            print(f"CNN 预测失败: {e}")
            return [0.0] * self.get_expected_dim()

    def _predict_sync(self, crop_images):
        """同步预测函数"""
        try:
            tensors = []
            for crop in crop_images:
                img_rgb = cv2.cvtColor(crop, cv2.COLOR_BGR2RGB)
                pil_img = Image.fromarray(img_rgb)
                tensors.append(self.transform(pil_img))

            if not tensors:
                return [0.0] * self.get_expected_dim()

            batch_tensor = torch.stack(tensors).to(DEVICE)
            with torch.no_grad():
                logits = self.cnn_model(batch_tensor)
                probs = torch.softmax(logits, dim=1)
                confs, preds = torch.max(probs, dim=1)

                preds = preds.cpu().numpy()
                confs = confs.cpu().numpy()

                adjusted_preds = []
                for i, (p, c) in enumerate(zip(preds, confs)):
                    if c < 0.6:
                        adjusted_preds.append(class_to_idx.get("none", 0))
                    else:
                        adjusted_preds.append(int(p))

                return self.convert_predictions_to_features(adjusted_preds)

        except Exception as e:
            print(f"同步预测失败: {e}")
            return [0.0] * self.get_expected_dim()

def normalize_tuple(x, y, w, h, W, H):
    return (round(x / W, 4), round(y / H, 4), round(w / W, 4), round(h / H, 4))


def get_health(crop, last):
    """获取血量百分比"""
    try:
        hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
        mask = cv2.inRange(hsv, HSV_LOW, HSV_HIGH)
        mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, KERNEL)
        row_counts = np.sum(mask[2:7] > 0, axis=1)
        if len(row_counts) == 0:
            return last
        min_val = int(np.min(row_counts))
        return round(max(0, min(min_val / 166.0, 1.0)) * 100, 4)
    except Exception as e:
        print(f"获取血量失败: {e}")
        return last

        self.last_step_time = 0
        self.min_step_interval = 0.01

        print(f"[Env {env_id}] 初始化完成 - 训练模式: {TRAINING_MODE}")
        self._game_started = False
        self._episode_done = False

    def check_human_input(self):
        """检查人类输入"""
        global TRAINING_MODE

        if self.env_id != 0:
            return False

        # 模式1才检测人类输入
        if TRAINING_MODE != 1:
            return False

        now = time.time()
        try:
            pressed = [int(keyboard.is_pressed(k)) for k in KEYS]
            if any(pressed):
                self.human_action = pressed
                self.last_human_time = now
                if not self.human_mode:
                    self.human_mode = True
                    if not self.in_demo_session:
                        self.data_manager.start_session()
                        self.in_demo_session = True
                        print("🎮 【人类示教】开始操作记录")
                return True
        except Exception as e:
            print(f"检查人类输入失败: {e}")
            return False

        if self.human_mode and (now - self.last_human_time > self.human_control_duration):
            self.human_mode = False
            if self.in_demo_session:
                self.data_manager.save_session()
                self.in_demo_session = False
                print("💾 【人类示教】操作记录已保存")
        return self.human_mode

    def _get_obs(self) -> Tuple[np.ndarray, float, float, float, set]:
        """获取观测数据"""
        global TRAINING_MODE

        # 离线模式返回模拟观测
        if TRAINING_MODE == 2:
            return self._get_simulated_obs()

        # FPS 控制
        now = time.time()
        elapsed = now - self.last_step_time
        if elapsed < self.min_step_interval:
            time.sleep(self.min_step_interval - elapsed)
        self.last_step_time = time.time()

        if self.sct is None:
            return self._get_default_obs()

        try:
            full_img = np.array(self.sct.grab(self.sct.monitors[0]))[..., :3]
        except Exception as e:
            print(f"屏幕捕获失败: {e}")
            return self._get_default_obs()

        # YOLO 检测
        yolo_flat, detected_classes = self._process_yolo(full_img)

        # 血量检测
        p1_blood, p2_blood = self._process_health(full_img)

        # CNN 处理
        cnn_vecs, time_val = self._process_cnn(full_img)

        # 更新血量分配
        self._update_blood_assignment(time_val, detected_classes, p1_blood, p2_blood)

        # 构造观测
        base_obs = [self.rblood, self.bblood] + yolo_flat
        obs = np.array(base_obs + cnn_vecs, dtype=np.float32)

        return obs, time_val, self.rblood, self.bblood, detected_classes

    def _get_simulated_obs(self):
        """离线模式：返回固定形状的全 0 观测"""
        obs = np.zeros(OBS_DIM, dtype=np.float32)
        return obs, -1.0, 1.0, 1.0, set()

    def _get_default_obs(self):
        """获取默认观测（当模型加载失败时）"""
        obs = np.zeros(OBS_DIM, dtype=np.float32)
        return obs, -1.0, 1.0, 1.0, set()

    def _process_yolo(self, full_img):
        """处理 YOLO 检测"""
        try:
            if self.yolo_model is None:
                return [0.0] * YOLO_DIM, set()

            g = GAME_REGION
            game_crop = full_img[g['top']:g['top'] + g['height'], g['left']:g['left'] + g['width']]
            results = self.yolo_model(cv2.cvtColor(game_crop, cv2.COLOR_BGR2RGB), size=320)

            class_to_idx_yolo = {'p': 0, 's': 1, 'r': 2, 'b': 3, 'wm': 4, 'rm': 5}
            boxes = {k: [] for k in TARGET_CLASSES}
            detected_classes = set()

            for *xyxy, conf, cls in results.pred[0]:
                cls_name = self.yolo_model.names[int(cls)]
                if cls_name in boxes:
                    detected_classes.add(cls_name)
                    x1, y1, x2, y2 = map(float, xyxy)
                    x_c, y_c = (x1 + x2) / 2, (y1 + y2) / 2
                    norm_box = normalize_tuple(x_c, y_c, x2 - x1, y2 - y1, g['width'], g['height'])
                    one_hot = [0.0] * 6
                    one_hot[class_to_idx_yolo[cls_name]] = float(conf)
                    boxes[cls_name].append(list(norm_box) + one_hot)

            yolo_flat = []
            for cls in sorted(TARGET_CLASSES):
                cls_boxes = boxes[cls][:4]
                while len(cls_boxes) < 4:
                    cls_boxes.append([0.0] * 10)
                for b in cls_boxes:
                    yolo_flat.extend(b)

            self._latest_boxes = boxes
            return yolo_flat, detected_classes
        except Exception as e:
            print(f"YOLO 处理失败: {e}")
            return [0.0] * YOLO_DIM, set()

    def _process_health(self, full_img):
        """处理血量检测"""
        try:
            crop1 = full_img[MONITOR_1P['top']:MONITOR_1P['top'] + MONITOR_1P['height'],
                    MONITOR_1P['left']:MONITOR_1P['left'] + MONITOR_1P['width']]
            crop2 = full_img[MONITOR_2P['top']:MONITOR_2P['top'] + MONITOR_2P['height'],
                    MONITOR_2P['left']:MONITOR_2P['left'] + MONITOR_2P['width']]

            p1_blood = get_health(crop1, self.last_health[0]) / 100.0
            p2_blood = get_health(crop2, self.last_health[1]) / 100.0
            self.last_health = [p1_blood * 100, p2_blood * 100]

            return p1_blood, p2_blood
        except Exception as e:
            print(f"血量检测失败: {e}")
            return 1.0, 1.0

    def _process_cnn(self, full_img):
        """处理 CNN 特征提取"""
        try:
            crops = []
            for reg in cnn_regions:
                crop = full_img[reg["top"]:reg["top"] + reg["height"], reg["left"]:reg["left"] + reg["width"]]
                crops.append(crop)

            cnn_vecs = self.cnn_processor.predict(crops)
            time_val = cnn_vecs[-1] if cnn_vecs else -1.0

            return cnn_vecs, time_val
        except Exception as e:
            print(f"CNN 处理失败: {e}")
            return [0.0] * CNN_DIM, -1.0

    def _update_blood_assignment(self, time_val, detected_classes, p1_blood, p2_blood):
        """更新血量分配：根据 b 的横坐标中点决定蓝方在左还是右"""
        if time_val >= 0.999:
            self.init_bblood = self.bblood
            self.init_rblood = self.rblood

            b_boxes = []
            for cls, boxes in self._latest_boxes.items():
                if cls == 'b':
                    b_boxes.extend(boxes)

            if b_boxes:
                avg_x = sum(b[0] for b in b_boxes) / len(b_boxes)
                pixel_x = avg_x * GAME_REGION['width']
                self.assign_flag = 'right' if pixel_x > 415 else 'left'
                if self.env_id == 0:
                    print(f"【判定】b 在 {'右' if self.assign_flag == 'right' else '左'}半屏")
            else:
                self.assign_flag = 'right'

        if self.assign_flag == 'right':
            self.bblood = p2_blood
            self.rblood = p1_blood
        else:
            self.bblood = p1_blood
            self.rblood = p2_blood

    def reset(self, *, seed=None, options=None):
        """重置环境：每次都强制等到下一轮倒计时 1.0"""
        super().reset(seed=seed)
        self._reset_state()
        self._episode_done = False

        # 关键：同步重置初始血量
        self.init_bblood = 1.0
        self.init_rblood = 1.0

        # 离线模式直接返回 0 观测
        if TRAINING_MODE == 2:
            return np.zeros(OBS_DIM, dtype=np.float32)

        self.start_counter = 0
        while True:
            obs, t, _, _, _ = self._get_obs()

            if abs(t - 1.0) < 0.01:
                self.start_counter += 1
            else:
                self.start_counter = 0

            if self.start_counter >= 5:
                if self.env_id == 0:
                    print("【提示】新一轮游戏开始")
                return obs, {}

            time.sleep(0.01)

    def _reset_state(self):
        """重置状态变量"""
        self.start_counter = 0
        self.end_counter_time = 0
        self.end_counter_p1 = 0
        self.end_counter_p2 = 0
        self.bblood = 1.0
        self.rblood = 1.0
        self.assign_flag = None
        self.last_obs = None
        self.last_action = None
        # 重置上一帧血量和奖励
        self.prev_bblood = self.bblood
        self.prev_rblood = self.rblood
        self.prev2_bblood = self.bblood
        self.prev2_rblood = self.rblood
        self.cumulative_reward = 0.0



    def step(self, action):
        """执行一步"""
        global TRAINING_MODE

        # 如果本局已结束，直接返回空奖励并保持终止状态
        if self._episode_done:
            obs = self._get_obs()[0]
            return obs, 0.0, True, False, {}

        # 记录上一步数据（供人类示教模式使用）
        if TRAINING_MODE == 1 and self.last_obs is not None:
            self.data_manager.add_experience(self.last_obs, self.last_action, 0.0,
                                             self._get_obs()[0], False)

        human_override = self.check_human_input()
        obs, t, p1, p2, _ = self._get_obs()

        # 保存当前 obs/action（下一步用）
        self.last_obs = obs.copy()
        self.last_action = action.copy()

        # 仅主环境执行按键
        if self.env_id == 0:
            if TRAINING_MODE == 1:  # 人类示教模式
                if human_override:
                    action = self.human_action
                self.release_all_keys()  # AI 不动手
            else:  # 正常/离线模式
                if human_override or self.human_mode:
                    if human_override:
                        action = self.human_action
                self._execute_actions(action)

        # 计算奖励
        reward = self._calculate_reward()

        # 检查游戏结束
        terminated = self._check_done(t, p1, p2)  # 真实终止
        truncated = False  # 无截断逻辑

        if terminated or truncated:
            self._episode_done = True
            if TRAINING_MODE == 1 and self.in_demo_session:
                self.data_manager.save_session()
                self.in_demo_session = False

        return obs, reward, terminated, truncated, {}

    def _execute_actions(self, action):
        """执行按键动作"""
        try:
            for i, pressed in enumerate(action):
                if pressed:
                    pyautogui.keyDown(KEYS[i])
                else:
                    pyautogui.keyUp(KEYS[i])
        except Exception as e:
            print(f"按键操作异常: {e}")

    def _calculate_reward(self):
        """基于血量差值的奖励（使用倒数第二帧 - 上一帧）"""
        # 计算差值：第二帧 - 上一帧
        deltab = (self.prev2_bblood - self.prev_bblood) * 100
        deltar = (self.prev2_rblood - self.prev_rblood) * 100

        # 强制置零：负值无效
        if deltab < 0:
            deltab = 0.0
        if deltar < 0:
            deltar = 0.0

        # 条件 1：双方血量都为 0 → 不更新
        if self.bblood == 0 and self.rblood == 0:
            # 滑动更新血量帧
            self.prev2_bblood, self.prev_bblood = self.prev_bblood, self.bblood
            self.prev2_rblood, self.prev_rblood = self.prev_rblood, self.rblood
            return self.cumulative_reward

        # 条件 2：回血 → 不更新
        if (self.prev_bblood < self.prev2_bblood and self.prev_bblood < self.bblood) or \
                (self.prev_rblood < self.prev2_rblood and self.prev_rblood < self.rblood):
            # 滑动更新血量帧
            self.prev2_bblood, self.prev_bblood = self.prev_bblood, self.bblood
            self.prev2_rblood, self.prev_rblood = self.prev_rblood, self.rblood
            return self.cumulative_reward

        # 新增条件：血量变化超过30% → 不更新（异常变化，忽略）
        if deltab > 30 or deltar > 30:
            # 滑动更新血量帧
            self.prev2_bblood, self.prev_bblood = self.prev_bblood, self.bblood
            self.prev2_rblood, self.prev_rblood = self.prev_rblood, self.rblood
            return self.cumulative_reward

        # 更新奖励
        old_reward = self.cumulative_reward
        self.cumulative_reward += (deltar - deltab)

        # 打印奖励更新
        if abs(self.cumulative_reward - old_reward) > 1e-6 and self.env_id == 0:
            print(f"reward: {old_reward:.1f}→{self.cumulative_reward:.1f} (Δb={deltab:.1f}, Δr={deltar:.1f})")

        # 滑动更新血量帧
        self.prev2_bblood, self.prev_bblood = self.prev_bblood, self.bblood
        self.prev2_rblood, self.prev_rblood = self.prev_rblood, self.rblood

        return self.cumulative_reward

    def _check_done(self, t, p1, p2):
        """检查游戏是否结束"""
        if abs(t) < 1e-3:
            self.end_counter_time += 1
        else:
            self.end_counter_time = 0

        if p1 <= 0:
            self.end_counter_p1 += 1
        else:
            self.end_counter_p1 = 0

        if p2 <= 0:
            self.end_counter_p2 += 1
        else:
            self.end_counter_p2 = 0

        done = (self.end_counter_time >= 5 or
                self.end_counter_p1 >= 5 or
                self.end_counter_p2 >= 5)

        if done:
            if self.env_id == 0:
                print("【环境】对局结束")
                self.release_all_keys()
            self.end_counter_time = self.end_counter_p1 = self.end_counter_p2 = 0

        return done

    def release_all_keys(self):
        """释放所有按键"""
        if self.env_id == 0 and not self.human_mode:
            try:
                for k in KEYS:
                    pyautogui.keyUp(k)
            except Exception as e:
                print(f"释放按键失败: {e}")

    def close(self):
        """关闭环境"""
        try:
            self.release_all_keys()
        except:
            pass
        if self.sct:
            try:
                self.sct.close()
            except:
                pass


# ============= 自定义策略相关 =============
class CustomExtractor(BaseFeaturesExtractor):
    def __init__(self, observation_space: spaces.Box, features_dim: int = 256):
        super().__init__(observation_space, features_dim)
        input_dim = observation_space.shape[0]
        self.net = nn.Sequential(
            nn.Linear(input_dim, 256), nn.ReLU(),
            nn.Linear(256, 256), nn.ReLU(),
            nn.Linear(256, features_dim)
        )

    def forward(self, obs):
        return self.net(obs)


policy_kwargs = dict(
    features_extractor_class=CustomExtractor,
    features_extractor_kwargs=dict(features_dim=256),
)


class BernoulliPolicy(ActorCriticPolicy):
    def _get_action_dist_from_latent(self, latent_pi: torch.Tensor):
        logits = self.action_net(latent_pi)
        torch_dist = TorchBernoulli(logits=logits)
        sb3_dist = BernoulliDistribution(torch.sigmoid(logits))
        sb3_dist.distribution = torch_dist
        return sb3_dist


class CustomPPO(PPO):
    def __init__(self, *args, imitation_weight: float = 0.1, **kwargs):
        super().__init__(*args, **kwargs)
        self.imitation_weight = imitation_weight

    def train(self) -> None:
        self.policy.train()
        clip_range = self.clip_range(self._current_progress_remaining) if callable(
            self.clip_range) else self.clip_range
        for _ in range(self.n_epochs):
            for rollout_data in self.rollout_buffer.get(self.batch_size):
                values, log_prob, entropy = self.policy.evaluate_actions(
                    rollout_data.observations, rollout_data.actions
                )
                adv = rollout_data.advantages
                if self.normalize_advantage:
                    adv = (adv - adv.mean()) / (adv.std() + 1e-8)
                ratio = torch.exp(log_prob - rollout_data.old_log_prob)
                surr1 = ratio * adv
                surr2 = torch.clamp(ratio, 1.0 - clip_range, 1.0 + clip_range) * adv
                policy_loss = -torch.min(surr1, surr2).mean()
                value_loss = F.mse_loss(rollout_data.returns, values.flatten())
                entropy_loss = -torch.mean(entropy)

                # 人类示范 loss
                imitation_loss = 0.0
                try:
                    if hasattr(self.env, "envs") and len(self.env.envs) > 0:
                        env0 = self.env.envs[0]
                        demo = getattr(env0, "data_manager", None)
                        if demo and demo.current_session:
                            k = min(len(demo.current_session), max(1, int(self.batch_size)))
                            sample = random.sample(demo.current_session, k)
                            obs_demo = torch.tensor([d['obs'] for d in sample], dtype=torch.float32,
                                                    device=self.device)
                            act_demo = torch.tensor([d['action'] for d in sample], dtype=torch.float32,
                                                    device=self.device)
                            dist = self.policy.get_distribution(obs_demo)
                            log_probs = dist.log_prob(act_demo).sum(dim=1)
                            imitation_loss = -torch.mean(log_probs)
                except Exception:
                    imitation_loss = 0.0

                loss = policy_loss + self.vf_coef * value_loss + self.ent_coef * entropy_loss + self.imitation_weight * imitation_loss
                self.policy.optimizer.zero_grad()
                loss.backward()
                if self.max_grad_norm is not None:
                    torch.nn.utils.clip_grad_norm_(self.policy.parameters(), self.max_grad_norm)
                self.policy.optimizer.step()
